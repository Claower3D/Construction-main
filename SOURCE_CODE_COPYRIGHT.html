257 for (const key of Object.keys(window)) {
258 if (!key.startsWith('AI_WRK_') && !key.startsWith('AI_WORK_')) continue;
259 const cat = window[key];
260 if (!cat || typeof cat !== 'object') continue;
261 for (const [, item] of Object.entries(cat)) {
262 if (!item || !item.name) continue;
263 if (pattern.test(item.name)) {
264 if (!unit || item.unit === unit) return item;
265 }
266 }

272 function findMaterial(searchText) {
273 const term = searchText.toLowerCase();
274 let best = null;
275 let bestScore = 0;
276 for (const key of Object.keys(window)) {
277 if (!key.startsWith('AI_MAT_')) continue;
278 const cat = window[key];
279 if (!cat || typeof cat !== 'object') continue;
280 for (const [, item] of Object.entries(cat)) {
281 if (!item || !item.name || !item.price) continue;
282 const name = item.name.toLowerCase();
283 const words = term.split(' ').filter(Boolean);
284 const matches = words.filter(w => name.includes(w)).length;
285 const score = matches / words.length;
286 if (score > bestScore) { bestScore = score; best = item; }
287 }
288 }
289 return bestScore >= 0.5 ? best : null;
290 }

301 for (const [, item] of Object.entries(cat)) {
302 if (!item || !item.name || !item.price) continue;
303 const name = item.name.toLowerCase();
304 const words = term.split(' ').filter(Boolean);
305 const matches = words.filter(w => name.includes(w)).length;
306 const score = matches / words.length;
307 if (score > bestScore) { bestScore = score; best = item; }
308 }
309 }
310 return bestScore >= 0.4 ? best : null;
311 }

315 // ═══════════════════════════════════════════════════════════
316 function resolveDimensions(qwenResult, objectParams, objectType) {
317 const qd = (qwenResult && qwenResult.dimensions_estimate) || {};
318 const p = objectParams || {};

321 const perimeter_m = qd.perimeter_m || (p.perimeter) || Math.sqrt(area_m2) * 4;

323 const depth_m = qd.depth_m || p.depth || 1.2;
324 const width_m = qd.width_m || p.width || 0.5;
325
326 // Производные

328 const rebar_kg = volume_m3 * 80; // ~80 кг/м³ для фундаментов
329 const rebar_t = rebar_kg / 1000;
330 const pile_count = Math.ceil(perimeter_m / 1.5);
331
332 return {
333 area_m2, perimeter_m, height_m, depth_m, width_m,
334 volume_m3, rebar_kg, rebar_t, pile_count,
335 };

341 function build({ objectType, qwenResult, objectParams, region, manualItems }) {
342 const type = objectType || 'generic';
343 const matrix = WORK_MATRIX[type] || WORK_MATRIX.generic;
344 const dims = resolveDimensions(qwenResult, objectParams, type);

396 addedByAI: false,

405 const materialsSection = [];
406 let totalMatsCost = 0;
407
408 for (const spec of matrix.materials) {
409 const mat = findMaterial(spec.search);
410 if (!mat) continue;
411

413 const wasteCoef = WASTE[spec.waste] || WASTE.default;
414 const qtyWithWaste = Math.ceil(qty * wasteCoef * 10) / 10;
415 const price = Math.round(mat.price * regCoef);
416 const subtotal = Math.round(qtyWithWaste * price);

433 addedByAI: false,
434 assumption: null,

461 stage: 'Оборудование',
462 description: `Аренда: ${eqSearch}`,
463 aiComment: null,
464 addedByAI: false,
465 });
466 }

486 const accuracy = qwenResult && qwenResult.confidence
487 ? Math.min(95, Math.round(qwenResult.confidence * 0.85 + 15))
488 : 65;
489
490 return {
491 objectType: type,
492 region: regionLabel,
493 dimensions: dims,
494 sections: {
495 works: worksSection,
496 materials: materialsSection,
497 equipment: equipSection,
498 labor: laborSection,
499 manuals: manuals,

501 totals: {
502 works: totalWorksCost,

504 equipment: totalEquipCost,
505 labor: laborSubtotal,
506 manual: manualTotal,
507 grand: grandTotal,
508 },
509 accuracy,
510 generatedAt: new Date().toISOString(),
511 };
512 }

394 /**
395 * Агрегирует несколько участков (patches) в единые параметры.
396 *
397 * @param {Array} patches - массив { area_m2, perimeter_m, height_m, ... }
398 * @param {string} mode - 'sum' | 'average' | 'max' | 'weighted_avg'
399 * @returns {Object} агрегированные параметры
400 */
401 function aggregatePatches(patches, mode = 'sum') {
402 if (!patches || !patches.length) return {};
403
404 if (patches.length === 1) return { ...patches[0], patch_count: 1 };
405
406 const result = {};
407 const numericKeys = new Set();
408 // Extensive: should always be summed
409 const sumKeys = new Set(['area_m2', 'perimeter_m', 'volume_m3', 'openings_area']);
410 // Intensive: should be averaged (weighted by area if available)
411 const avgKeys = new Set(['height_m', 'width_m', 'depth_m', 'confidence',
412 'thickness_m', 'foundation_depth', 'foundation_width',
413 'screed_thickness', 'slab_thickness']);
414
415 // Collect all numeric keys
416 for (const p of patches) {
417 for (const [k, v] of Object.entries(p)) {
418 if (typeof v === 'number') numericKeys.add(k);
419 }
420 }
421
422 // Total area for weighting
423 const totalArea = patches.reduce((s, p) => s + (p.area_m2 || 0), 0) || 1;
424
425 for (const key of numericKeys) {
426 const values = patches.map(p => p[key] || 0);
427
428 if (mode === 'weighted_avg') {
429 if (sumKeys.has(key)) {
430 result[key] = values.reduce((a, b) => a + b, 0);
431 } else if (avgKeys.has(key)) {
432 // Area-weighted average
433 let wSum = 0;
434 for (let i = 0; i < patches.length; i++) {
435 const w = (patches[i].area_m2 || 1) / totalArea;
436 wSum += (patches[i][key] || 0) * w;
437 }
438 result[key] = wSum;
439 } else {
440 result[key] = values.reduce((a, b) => a + b, 0);
441 }
442 } else {
443 switch (mode) {
444 case 'sum':
445 result[key] = values.reduce((a, b) => a + b, 0);
446 break;
447 case 'average':
448 result[key] = values.reduce((a, b) => a + b, 0) / values.length;
449 break;
450 case 'max':
451 result[key] = Math.max(...values);
452 break;
453 }
454 }
455 }
456
457 // Preserve non-numeric fields from first patch
458 const first = patches[0];
459 for (const [k, v] of Object.entries(first)) {
460 if (typeof v !== 'number' && result[k] === undefined) {
461 result[k] = v;
462 }
463 }
464
465 result.patch_count = patches.length;
466 result.total_area = patches.reduce((s, p) => s + (p.area_m2 || 0), 0);
467 result.total_volume = patches.reduce((s, p) => s + (p.volume_m3 || 0), 0);
468 return result;
469 }

99 /**
100 * Check if transition is allowed
101 * @param {string} entityType - 'Order', 'Application', 'Work', 'Stage', 'Defect'
102 * @param {string} fromStatus - current status
103 * @param {string} toStatus - target status
104 * @returns {boolean}
105 */
106 function canTransition(entityType, fromStatus, toStatus) {
107 const rules = TransitionRules[entityType];
108 if (!rules) {
109 console.error(`Unknown entity type: ${entityType}`);
110 return false;
111 }
112
113 const allowed = rules[fromStatus];
114 if (!allowed) {
115 console.error(`Unknown status: ${fromStatus} for ${entityType}`);
116 return false;
117 }
118
119 return allowed.includes(toStatus);
120 }
121
122 /**
123 * Get allowed transitions from current status
124 * @param {string} entityType
125 * @param {string} currentStatus
126 * @returns {string[]} array of allowed target statuses
127 */
128 function getAllowedTransitions(entityType, currentStatus) {
129 const rules = TransitionRules[entityType];
130 if (!rules) return [];
131 return rules[currentStatus] || [];
132 }

134 /**
135 * Perform status transition with validation and audit logging
136 * @param {object} entity - entity object with id and status
137 * @param {string} entityType - 'Order', 'Application', etc.
138 * @param {string} toStatus - target status
139 * @param {object} actor - user performing the action { id, role, name }
140 * @param {object} meta - additional metadata
141 * @returns {object} { success: boolean, error?: string }
142 */
143 function transition(entity, entityType, toStatus, actor = {}, meta = {}) {
144 const fromStatus = entity.status;
145
146 // Check if transition is allowed
147 if (!canTransition(entityType, fromStatus, toStatus)) {
148 const error = `Переход ${fromStatus} → ${toStatus} запрещён для ${entityType}`;
149 console.error(error);
150 return { success: false, error };
151 }
152
153 // Update entity status
154 entity.status = toStatus;
155 entity.updatedAt = new Date().toISOString();
156
157 // Add timestamp for specific status changes
158 if (entityType === 'Order') {
159 if (toStatus === OrderStatus.PUBLISHED) entity.publishedAt = new Date().toISOString();
160 if (toStatus === OrderStatus.IN_WORK) entity.startedAt = new Date().toISOString();
161 if (toStatus === OrderStatus.DONE) entity.completedAt = new Date().toISOString();
162 }
163
164 if (entityType === 'Work') {
165 if (toStatus === WorkStatus.ON_REVIEW) entity.submittedAt = new Date().toISOString();
166 if (toStatus === WorkStatus.DONE) entity.completedAt = new Date().toISOString();
167 }
168
169 if (entityType === 'Defect') {
170 if (toStatus === DefectStatus.FIXED) entity.fixedAt = new Date().toISOString();
171 if (toStatus === DefectStatus.CONFIRMED) entity.confirmedAt = new Date().toISOString();
172 }
173
174 // Save entity
175 entity.save();
176
177 // Create audit log
178 AuditLog.log(entityType.toLowerCase(), entity.id, 'statusChanged', {
179 fromStatus,
180 toStatus,
181 meta
182 });
183
184 console.log(`✅ ${entityType} ${entity.id}: ${fromStatus} → ${toStatus}`);
185 return { success: true };
186 }

661 ruleCode: deadline.ruleCode,
662 message: rule.message,
663 action: 'auto_accept'
664 });
665 return { success: true, action: 'auto_accept' };
666 }
667

669 AuditLog.log(rule.entityType.toLowerCase(), deadline.entityId, 'deadline_expired', {
670 ruleCode: deadline.ruleCode,
671 message: rule.message,
672 action: 'notify_warning'
673 });
674 return { success: true, action: 'notify_warning' };
675 }

678 AuditLog.log(rule.entityType.toLowerCase(), deadline.entityId, 'deadline_escalated', {
679 ruleCode: deadline.ruleCode,
680 message: rule.message,
681 action: 'escalate'
682 });

639 @staticmethod
640 def _iou(box1: tuple, box2: tuple) -> float:
641 x1 = max(box1[0], box2[0])
642 y1 = max(box1[1], box2[1])
643 x2 = min(box1[2], box2[2])
644 y2 = min(box1[3], box2[3])
645 inter = max(0, x2 - x1) * max(0, y2 - y1)
646 area1 = (box1[2] - box1[0]) * (box1[3] - box1[1])
647 area2 = (box2[2] - box2[0]) * (box2[3] - box2[1])
648 union = area1 + area2 - inter

479 def _calculate_quantity(
480 self,
481 detection,
482 unit: str,
483 multiplier: float
484 ) -> float:
485 """Calculate quantity based on unit type and detection measurements."""
486
487 if unit == "m3":
488 # Volume
489 if detection.volume_m3:
490 return detection.volume_m3 * multiplier
491 elif detection.area_m2 and detection.depth_m:
492 return detection.area_m2 * detection.depth_m * multiplier
493 elif detection.area_m2:
494 return detection.area_m2 * 1.0 * multiplier # Assume 1m depth
495 else:
496 return 1.0 * multiplier # Fallback
497
498 elif unit == "m2":
499 # Area
500 if detection.area_m2:
501 return detection.area_m2 * multiplier
502 elif detection.width_m and detection.height_m:
503 return detection.width_m * detection.height_m * multiplier
504 else:
505 return 1.0 * multiplier
506
507 elif unit == "m":
508 # Length (use max of width/height)
509 if detection.width_m or detection.height_m:
510 return max(detection.width_m or 0, detection.height_m or 0) * multiplier
511 else:
512 return 1.0 * multiplier
513
514 elif unit == "kg":
515 # Weight (typically for rebar, based on volume)
516 if detection.volume_m3:
517 return detection.volume_m3 * multiplier # multiplier = kg/m3
518 else:
519 return 10.0 * multiplier # Fallback
520
521 elif unit == "pcs":
522 # Pieces (count)
523 return 1.0 * multiplier
524
525 else:
526 return 1.0 * multiplier

280 let _maxContextLength = 20;

282 // ─── CORE: PROCESS USER MESSAGE ──────────────────────
283 // ПРИОРИТЕТ: Gemini LLM → Rule-based fallback
284 // Gemini понимает человеческую речь, логику, контекст.

287 async function processMessage(userMessage, opts = {}) {
288 const msg = userMessage.toLowerCase().trim();
289
290 // Сохраняем контекст разговора
291 _conversationContext.push({ role: 'user', text: userMessage, time: Date.now() });
292 if (_conversationContext.length > _maxContextLength) {
293 _conversationContext.shift();
294 }
295
296 // Определяем контекст (для обогащения промпта Gemini)
297 const matchedPage = _findMatchingPage(msg);
298 const matchedWorkflow = _findMatchingWorkflow(msg);
299
300 let response = null;
301
302 // ═══ ШАГA 1: ВСЕГДА сначала Gemini LLM ═══
303 // Gemini понимает: разговорный язык, логику, контекст,
304 // неполные фразы, опечатки, сленг, сарказм, всё.
305 try {
306 response = await _askGemini(userMessage, matchedPage, opts);
307 } catch (e) {
308 console.warn('[SmartBrain] Gemini unavailable:', e.message);
309 }
310
311 // ═══ ШАГ 2: Если Gemini недоступен → Rule-based fallback ═══
312 if (!response) {
313 const intent = _detectIntent(msg);
314 if (intent) {
315 const ruleResponse = await _executeIntent(intent, msg, matchedPage, matchedWorkflow, opts);
316 if (ruleResponse && !ruleResponse.useLLM) {
317 response = ruleResponse;
318 response.source = 'rules';
319 }
320 }
321 }
322
323 // ═══ ШАГ 3: Последний fallback — умная подсказка ═══
324 if (!response) {
325 response = _buildSmartFallback(msg, matchedPage);
326 response.source = 'fallback';
327 }
328
329 // Сохраняем контекст ответа
330 _conversationContext.push({ role: 'bot', text: response.text, time: Date.now() });
331
332 return response;
333 }

337 function _detectIntent(msg) {
338 let bestIntent = null;
339 let bestScore = 0;
340
341 for (const [intentName, config] of Object.entries(INTENTS)) {
342 for (const pattern of config.patterns) {
343 if (msg.includes(pattern)) {
344 const score = pattern.length;
345 if (score > bestScore) {
346 bestScore = score;
347 bestIntent = { name: intentName, pattern, handler: config.handler };
348 }
349 }
350 }
351 }
352
353 return bestIntent;
354 }

358 function _findMatchingPage(msg) {
359 let bestMatch = null;
360 let bestScore = 0;
361
362 for (const [pageId, page] of Object.entries(APP_MAP.pages)) {
363 for (const keyword of page.keywords) {
364 if (msg.includes(keyword.toLowerCase())) {
365 const score = keyword.length;
366 if (score > bestScore) {
367 bestScore = score;
368 bestMatch = page;
369 }
370 }
371 }
372 }
373
374 return bestMatch;

800 function _extractActionsFromLLM(text) {
801 const actions = [];
802 const lower = text.toLowerCase();
803
804 // Ищем упомянутые страницы
805 for (const [pageId, page] of Object.entries(APP_MAP.pages)) {
806 const pageName = page.name.toLowerCase();
807 if (lower.includes(pageName) || lower.includes(pageId)) {
808 actions.push({
809 text: `${page.icon} Открыть ${page.name}`,
810 page: pageId
811 });
812 if (actions.length >= 3) break; // максимум 3 действия
813 }
814 }
815
816 // Если ответ Gemini содержит цены/сметы → добавляем кнопку "Рассчитать смету"
817 const hasEstimateContext = /(\d[\d\s]*₸|стоимост|смет|расчёт|цен[аы]|за м[²2]|за метр)/i.test(text);
818 const userMsg = _conversationContext.filter(m => m.role === 'user').pop();
819 const userMsgText = userMsg ? userMsg.text : '';
820 const isConstructionQuery = /ремонт|штукатур|плитк|кровл|фундамент|электрик|сантехник|стяжк|покраск|утеплен|фасад|демонтаж|кладк|стен|пол|потол|кухн|ванн|балкон|крыш/i.test(userMsgText);
821
822 if (hasEstimateContext && isConstructionQuery) {
823 // Не дублируем если уже есть кнопка Оценки
824 const alreadyHasEstimate = actions.some(a => a.page === 'estimates');
825 if (!alreadyHasEstimate) {
826 actions.push({
827 text: '📊 Рассчитать полную смету',
828 page: 'estimates',
829 prefillDescription: userMsgText
830 });
831 }
832 }
833
834 return actions.length > 0 ? actions : null;
835 }
